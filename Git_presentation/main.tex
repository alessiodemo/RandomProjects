\begin{frame}{Git: Version Control System}
  \vspace{0.03cm}
  \setlength{\columnsep}{1.2cm}
  \begin{columns}[T,onlytextwidth]
    \column{0.36\textwidth}
      \scriptsize
      \textbf{What is it?}
      \begin{itemize}
        \item Git tracks project changes over time.
        \item It helps teams collaborate without losing work.
        \item Every saved change is a \textbf{commit}.
      \end{itemize}

      \vspace{0.10cm}
      \begin{block}{Why use it?}
        \scriptsize
        \begin{itemize}
          \item Recover previous versions
          \item Work on multiple features
          \item Collaborate safely
        \end{itemize}
      \end{block}

    \column{0.64\textwidth}
      \centering
      \vspace{0.04cm}
      \resizebox{0.62\linewidth}{!}{%
      \begin{tikzpicture}[
        box/.style={draw, rounded corners, fill=gray!20, minimum width=2.0cm, minimum height=0.58cm, align=center, font=\tiny},
        labelbox/.style={fill=gray!12, inner sep=2pt, font=\tiny},
        arrow/.style={->, thick, >=Stealth, gray!70}
      ]
        \node[box] (wd) {Working Directory};
        \node[box, below=0.72cm of wd] (stage) {Staging Area};
        \node[box, below=0.72cm of stage] (repo) {Local Repo};
        \node[box, below=0.72cm of repo] (remote) {Remote Repo};

        \draw[arrow] (wd) -- node[right, labelbox]{git add} (stage);
        \draw[arrow] (stage) -- node[right, labelbox]{git commit} (repo);
        \draw[arrow, red!70] ([xshift=5pt]repo.east) -- node[right, labelbox]{git push} ([xshift=5pt]remote.east);
        \draw[arrow, blue!70] ([xshift=-5pt]remote.west) -- node[left, labelbox]{git pull} ([xshift=-5pt]repo.west);
      \end{tikzpicture}%
      }
    \end{columns}

    \vspace{0.08cm}

      \texttt{git pull} \quad \texttt{git push}
    \end{block}
