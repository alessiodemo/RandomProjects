\begin{frame}{What is Git?}
  \begin{itemize}
    \item Git is a distributed version control system.
    \item It tracks project changes over time.
    \item It helps teams collaborate without losing work.
    \item Every saved update is a \textbf{commit}.
  \end{itemize}

  \vspace{0.1cm}
  \begin{center}
    \begin{tikzpicture}[
      box/.style={draw, rounded corners, fill=gray!20, minimum width=1.9cm, minimum height=0.68cm, align=center, font=\tiny},
      arrow/.style={->, thick, >=Stealth, gray!70}
    ]
      \node[box] (wd) {Working Directory};
      \node[box, right=1.1cm of wd] (stage) {Staging Area};
      \node[box, right=1.1cm of stage] (repo) {Local Repository};
      \node[box, below=0.9cm of repo] (remote) {Remote Repository};

      \draw[arrow] (wd) -- node[above, font=\tiny]{git add} (stage);
      \draw[arrow] (stage) -- node[above, font=\tiny]{git commit} (repo);
      \draw[arrow] (repo) -- node[right, font=\tiny]{git push} (remote);
      \draw[arrow] (remote) -- node[left, font=\tiny]{git pull} (repo);
    \end{tikzpicture}
  \end{center}

  \vspace{0.1cm}
  \begin{block}{Why use it?}
    \begin{itemize}
      \item Recover previous versions
      \item Work on multiple features
      \item Collaborate safely
    \end{itemize}
  \end{block}
\end{frame}

\begin{frame}[fragile]{Main commands}
  \scriptsize
  \begin{lstlisting}[language=bash]
# initialize repository
$ git init

# check status
$ git status

# add files
$ git add file_name

# save a snapshot
$ git commit -m "message"

# create a branch
$ git checkout -b new_feature

# update local repo from remote
$ git pull origin main
  \end{lstlisting}

  \vspace{0.08cm}
  \begin{alertblock}{Key idea}
    \small
    \begin{minipage}{0.9\textwidth}
      \textbf{git push} sends local changes to the remote repository.\newline
      \textbf{git pull} updates your local copy from the remote repository.
    \end{minipage}
  \end{alertblock}
\end{frame}
